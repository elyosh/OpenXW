option(XW_REQUIRE_MIDI_BACKENDS "Require FluidSynth and Nuked SC-55 for distributable packages" OFF)
if(XW_REQUIRE_MIDI_BACKENDS)
    get_target_property(imuse_definitions imuse COMPILE_DEFINITIONS)
    foreach(backend IN ITEMS HAVE_FLUIDSYNTH HAVE_NUKED_SC55)
        if(NOT "${backend}=1" IN_LIST imuse_definitions)
            message(FATAL_ERROR "Package requires ${backend}: install FluidSynth and SpeexDSP and enable their iMUSE backends")
        endif()
    endforeach()
endif()
